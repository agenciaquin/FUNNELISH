# Replicar SOLO la estructura de EMBUDOS (constructor + bloques + página de venta + checkout)

Esto es el subconjunto para llevar **únicamente el creador de embudos** a otra app,
sin el bot de WhatsApp, ni chat, ni Effi, ni vendedores, ni Meta Ads.

Incluye: el **editor** (lista + constructor con teléfono), **todos los bloques**, la
**página de venta** pública, el **checkout** (productos, colores, tallas, packs),
carritos abandonados y estadísticas del embudo.

## 2. Base de datos (solo lo del embudo)

Pega en Supabase → SQL Editor. Idempotente. (NO necesitas conversations/messages.)

```sql
-- EMBUDOS
create table if not exists funnels (
  id uuid primary key default gen_random_uuid(),
  slug text not null unique, activo boolean default true,
  nombre text, titulo text, producto text, precio numeric, precio_antes numeric,
  imagenes jsonb default '[]'::jsonb,
  imagen_banner text, imagen_clientes text, imagen_detalle text,
  caracteristicas jsonb default '[]'::jsonb, tallas jsonb default '[]'::jsonb,
  variantes jsonb default '[]'::jsonb, frases jsonb default '[]'::jsonb,
  horas_contador integer default 10, personas_comprando integer default 27,
  whatsapp text, pixel_meta text, pixel_meta_token text, pixel_tiktok text, pixel_tiktok_token text,
  audio_url text, video_url text, color text, miniatura_url text, anuncios text,
  layout jsonb, insignia jsonb,
  eliminado boolean not null default false, eliminado_at timestamptz,
  creado_at timestamptz default now()
);

-- PEDIDOS (versión mínima para el checkout)
create table if not exists clientes_funnelish (
  id bigint generated by default as identity primary key,
  telefono text not null, nombre text, producto text, talla text, valor text,
  ciudad text, departamento text, direccion text, correo text,
  referencia text, estado text default 'pendiente', confirmado boolean default false,
  cantidad integer not null default 1, foto_producto text, funnel_slug text,
  utm_source text, utm_medium text, utm_campaign text, referrer text,
  abono numeric default 0, abono_recibido boolean default false,
  created_at timestamptz default now(), updated_at timestamptz default now()
);
create index if not exists idx_cf_funnel_slug on clientes_funnelish(funnel_slug);

-- CARRITOS ABANDONADOS
create table if not exists carritos_abandonados (
  id uuid primary key default gen_random_uuid(),
  slug text not null, telefono text not null,
  nombre text, producto text, talla text, valor numeric,
  datos jsonb, nota text, recuperado boolean not null default false, recuperado_at timestamptz,
  created_at timestamptz not null default now(), updated_at timestamptz not null default now()
);
create unique index if not exists uq_carrito_slug_tel on carritos_abandonados(slug, telefono);

-- RASTREO
create table if not exists funnel_eventos (
  id uuid primary key default gen_random_uuid(),
  slug text not null, paso text not null,
  utm_source text, utm_medium text, utm_campaign text, referrer text,
  created_at timestamptz not null default now()
);
create index if not exists idx_fe_slug_paso on funnel_eventos(slug, paso, created_at);

-- CATÁLOGO DE COLORES (para "Traer de catálogo"; opcional)
create table if not exists catalogos (
  id uuid primary key default gen_random_uuid(), familia text, patron text, anuncios text,
  created_at timestamptz default now()
);
create table if not exists catalogo_colores (
  id uuid primary key default gen_random_uuid(),
  catalogo_id uuid references catalogos(id) on delete cascade,
  color text not null, nombre_producto text not null, url_imagen text, url_original text,
  variante jsonb default '{}'::jsonb, activo boolean default true, orden integer default 0,
  created_at timestamptz default now()
);

-- PERMISOS (el servidor usa la service role key)
grant all on all tables in schema public to service_role;
grant all on all sequences in schema public to service_role;
alter default privileges in schema public grant all on tables to service_role;
alter default privileges in schema public grant all on sequences to service_role;
```

## 4. El único ajuste obligatorio: guardar el pedido

En el proyecto original, `app/api/pedidos/route.ts` reenvía al webhook del bot de WhatsApp
(que además manda mensajes, arma collage, etc.). Como aquí **NO** quieres el bot,
reemplaza ese endpoint por uno que **solo inserte el pedido** en `clientes_funnelish`:

```ts
// app/api/pedidos/route.ts (versión mínima — solo guarda)
import { NextRequest, NextResponse } from 'next/server';
import { createServerSupabaseClient } from '@/lib/supabase';

export async function POST(req: NextRequest) {
  const b = await req.json();
  const req0 = ['nombre','apellidos','whatsapp','direccion','barrio','municipio','departamento'];
  for (const c of req0) if (!String(b?.[c] ?? '').trim())
    return NextResponse.json({ error: `Falta ${c}` }, { status: 400 });

  const tel = String(b.whatsapp).replace(/\D/g,'').replace(/^57/,'');
  if (!/^3\d{9}$/.test(tel)) return NextResponse.json({ error: 'WhatsApp inválido' }, { status: 400 });

  const referencia = String(b.referencia ?? '').trim() || `web-${Date.now()}`;
  const supabase = createServerSupabaseClient();
  const { error } = await supabase.from('clientes_funnelish').insert({
    telefono: tel, nombre: `${b.nombre} ${b.apellidos}`.trim(),
    producto: b.variante ?? null, talla: b.talla ?? null, valor: String(b.precio ?? ''),
    direccion: `${b.direccion}, ${b.barrio}`.trim(), ciudad: b.municipio ?? null,
    departamento: b.departamento ?? null, correo: b.correo ?? null,
    referencia, estado: 'pendiente', foto_producto: b.imagen ?? null,
    funnel_slug: String(b.slug ?? '').trim() || null,
    utm_source: b.utms?.utm_source ?? null, utm_campaign: b.utms?.utm_campaign ?? null,
    referrer: b.referrer ?? null,
  });
  if (error) return NextResponse.json({ error: 'No se pudo registrar' }, { status: 500 });
  return NextResponse.json({ ok: true, referencia });
}
```

Todo lo demás (checkout, bloques, editor, carritos, stats) funciona igual con lo copiado.
